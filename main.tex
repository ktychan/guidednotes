\begin{luacode*}
local function get_aux_dir()
  for _, v in ipairs(arg or {}) do
    local d = v:match("%-%-?output%-directory=(.+)")
               or v:match("%-%-?aux%-directory=(.+)")
    if d then return d end
  end
  return "."
end

local auxdir = get_aux_dir()
local offset = titlepage + frontmatter
local prev   = offset + 1

local csv = io.open(auxdir .. "/weeks.csv", "w")
csv:write("i,a,b\n")
for i, pagenum in ipairs(weeks) do
  local a = prev
  local b = pagenum + offset - 1
  csv:write(string.format("%d,%d,%d\n", i, a, b))
  prev = b + 1
end
csv:close()
